% EML Whitepaper — Custom LaTeX Preamble
% Math macros for consistent notation throughout the paper.

% ─── Two-column overflow mitigation ──────────────────────────────
\sloppy                              % permit looser word spacing
\setlength{\emergencystretch}{3em}   % extra stretch before overflow
\allowdisplaybreaks                  % allow page breaks in display math

% ─── Algorithm & Hash ─────────────────────────────────────────────

\newcommand{\nulla}{\bot_a}                  % null constant for algorithm a
\newcommand{\nulltable}{\mathit{NullTable}}  % precomputed null table
\newcommand{\proj}{\pi_a}                    % projection for algorithm a
\newcommand{\act}{\mathit{act}}              % activation map
\newcommand{\leafval}{V}                     % leaf value function
\newcommand{\treesz}[1]{\mathit{tree\_size}(#1)}

% ─── Null constants at height h ───────────────────────────────────
\newcommand{\Nh}[1]{N_{#1}}                  % N_h(a) — null at height h

% ─── Equational law names ─────────────────────────────────────────

% ─── Complexity ───────────────────────────────────────────────────

% ─── Misc ─────────────────────────────────────────────────────────
\newcommand{\concat}{\mathbin{\Vert}}        % concatenation operator
\newcommand{\eml}{\textsc{eml}}              % small-caps EML

% ─── Suppress amsthm auto-numbering ──────────────────────────────
% Quarto generates \newtheorem{definition}{Definition}[section] and
% also writes "Definition N" in the prose body, causing duplication
% (e.g. "Definition 4.3. Definition 3"). We define a headless theorem
% style and redefine both environments to use it, so only Quarto's
% own label appears.
\newtheoremstyle{headless}%
  {3pt}{3pt}{\itshape}{}{}{}{ }{\relax}%
\AtBeginDocument{%
  \theoremstyle{headless}%
  \let\definition\relax
  \let\enddefinition\relax
  \let\c@definition\relax
  \newtheorem*{definition}{}%
  \let\theorem\relax
  \let\endtheorem\relax
  \let\c@theorem\relax
  \newtheorem*{theorem}{}%
}
